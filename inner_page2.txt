<script>
        // Simple script to handle toggle visuals for demonstration
        document.querySelectorAll('button[role="switch"]').forEach(toggle => {
            toggle.addEventListener('click', (e) => {
                e.stopPropagation(); // prevent parent click
                const isChecked = toggle.getAttribute('aria-checked') === 'true';
                const thumb = toggle.querySelector('span');
                
                if (isChecked) {
                    toggle.setAttribute('aria-checked', 'false');
                    toggle.classList.replace('bg-secondary', 'bg-surface-container-highest');
                    thumb.classList.replace('translate-x-5', 'translate-x-0');
                    thumb.classList.replace('bg-on-secondary', 'bg-outline-variant');
                } else {
                    toggle.setAttribute('aria-checked', 'true');
                    toggle.classList.replace('bg-surface-container-highest', 'bg-secondary');
                    thumb.classList.replace('translate-x-0', 'translate-x-5');
                    thumb.classList.replace('bg-outline-variant', 'bg-on-secondary');
                }
            });
        });
    </script>
